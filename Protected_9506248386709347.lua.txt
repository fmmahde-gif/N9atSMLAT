local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)

    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

-- =========================================================
-- Remove Player Entry
-- =========================================================

local function removePlayerEntry(player)

    -- Protected users are never stored
    if isProtected(player) then
        return
    end

    local userId = player.UserId
    local data = playerData[userId]

    if not data then
        return
    end

    data.Exited = data.Exited + 1

    if data.JoinTime then
        data.TotalTime =
            data.TotalTime + tick() - data.JoinTime
    end

    data.JoinTime = nil

    if data.Instance then
        data.Instance:Destroy()
    end

    playerData[userId] = nil

    sound:Play()
end

-- =========================================================
-- Register Player
-- =========================================================

local function registerPlayer(player)

    if isProtected(player) then
        return
    end

    local userId = player.UserId
    local data = playerData[userId]

    if not data then
        return
    end

    if data.JoinTime then
        data.Entered = data.Entered + 1
    end

    data.JoinTime = tick()

    sound:Play()
end

-- Refresh Players
-- =========================================================

local function refreshPlayers()

    local query = string.lower(searchBox.Text)

    for _, player in ipairs(Players:GetPlayers()) do

        -- Protected users are ignored completely
        if not isProtected(player) then

            local name = string.lower(player.Name)

            if query == "" or name:find(query, 1, true) then

                if not playerData[player.UserId] then
                    createPlayerEntry(player)
                end

            end
        end
    end
end

local function sendMahdiMessage()

    pcall(function()

        local textChannels =
            TextChatService:FindFirstChild("TextChannels")

        if not textChannels then
            return
        end

        local generalChannel =
            textChannels:FindFirstChild("RBXGeneral")

        if generalChannel then
            generalChannel:SendAsync("عمي مهدي")
        end

    end)
end

local function processChatMessage(message)

    if typeof(message) ~= "string" then
        return
    end

    -- Remove spaces
    local cleaned =
        message:gsub("%s+", ""):lower()

    if cleaned == "?3mk" then
        sendMahdiMessage()
    end
end

-- Modern Roblox chat
pcall(function()

    TextChatService.MessageReceived:Connect(function(message)

        if message and message.TextSource then
            processChatMessage(message.Text)
        end

    end)

end)

-- Legacy chat support
pcall(function()

    LocalPlayer.Chatted:Connect(function(message)
        processChatMessage(message)
    end)

end)

                    total + tick() - data.JoinTime

            local player = data.Player

            if player then

                data.Instance.InfoLabel.Text =
                    string.format(
                        "%s: مدة البقاء %s  دخول %d  خروج %d",
                        player.Name,
